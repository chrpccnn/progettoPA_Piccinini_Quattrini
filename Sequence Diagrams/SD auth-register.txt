title Sequence diagram - POST /auth/register

client->authRoutes: POST /auth/register

authRoutes->authValidatorMiddleware: authSchema.parse(req.body)

alt invalid authentication data
    authValidatorMiddleware->errorHandler: next(BadRequest)
    errorHandler-->client: 400 Invalid authentication data
else valid authentication data
    authValidatorMiddleware-->authRoutes: next()

    authRoutes->authController: register(req, res, next)
    authController->authService: register(email, password)

    authService->authRepository: findByEmail(email)
    authRepository->MySQL: SELECT user WHERE email = ?
    MySQL-->authRepository: User | undefined
    authRepository-->authService: User | undefined

    alt user already exists
        authService-->authController: throw UserAlreadyExists
        authController->errorHandler: next(err)
        errorHandler-->client: 409 Email already exists
    else user does not exist
        authService->authService: bcrypt.hash(password, 10)

        authService->authRepository: createNewUser(email, passwordHash)
        authRepository->MySQL: INSERT user with role = "user"
        MySQL-->authRepository: User
        authRepository-->authService: User

        authService-->authController: User data
        authController-->client: 201 Created + JSON
    end
end
