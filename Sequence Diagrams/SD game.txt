title Sequence diagram - POST /game

client->gameRoutes: POST /game

gameRoutes->authMiddleware: verify JWT
authMiddleware-->gameRoutes: next()

gameRoutes->roleMiddleware: requireRole("user")
roleMiddleware-->gameRoutes: next()

gameRoutes->gameValidatorMiddleware: validate game body with Zod
gameValidatorMiddleware-->gameRoutes: next()

gameRoutes->gameController: createGame(req, res, next)
gameController->gameService: createGame(userId, body)

gameService->gameService: begin transaction

alt mode == "user"
    gameService->UserRepository: findByEmail(opponentEmail)
    UserRepository->MySQL: SELECT opponent
    MySQL-->UserRepository: User | null
    UserRepository-->gameService: opponent

    alt opponent not found
        gameService-->gameController: throw UserNotFound
        gameController->errorHandler: next(err)
        errorHandler-->client: 404 User not found
    end
end

gameService->UserRepository: lockByIds(userIds, transaction)
UserRepository->MySQL: SELECT users FOR UPDATE
MySQL-->UserRepository: locked users
UserRepository-->gameService: users

gameService->gameService: verify creator TOKEN balance

gameService->gameRepository: hasActiveGame(userId, transaction)
gameRepository->MySQL: check active game
MySQL-->gameRepository: true | false
gameRepository-->gameService: result

alt creator already in active game
    gameService-->gameController: throw UserAlreadyInGame
    gameController->errorHandler: next(err)
    errorHandler-->client: 409 Conflict
end

alt mode == "user"
    gameService->gameRepository: hasActiveGame(opponentId, transaction)
    gameRepository->MySQL: check opponent active game
    MySQL-->gameRepository: true | false
    gameRepository-->gameService: result

    alt opponent already in active game
        gameService-->gameController: throw OpponentAlreadyInGame
        gameController->errorHandler: next(err)
        errorHandler-->client: 409 Conflict
    end
end

gameService->GameDomain: create Grid and random Fleets
GameDomain-->gameService: Grid + Fleets

alt mode == "user"
    gameService->GameDomain: create HumanPlayers and Game
else mode == "ai"
    gameService->GameDomain: create HumanPlayer, AIPlayer and Game
end

GameDomain-->gameService: Game

gameService->GameDomain: payCreationCost()
gameService->GameDomain: start()

gameService->gameRepository: insert(game, transaction)
gameRepository->MySQL: INSERT game, players, ships, positions, grid and TOKEN changes
MySQL-->gameRepository: persisted
gameRepository-->gameService: saved

gameService->gameService: commit transaction
gameService-->gameController: create game result
gameController-->client: 201 Created + JSON
