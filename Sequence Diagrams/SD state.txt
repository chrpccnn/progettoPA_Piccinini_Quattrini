title Sequence diagram - GET /game/:gameId/state

client->gameRoutes: GET /game/:gameId/state

gameRoutes->authMiddleware: verify JWT
authMiddleware-->gameRoutes: next()

gameRoutes->roleMiddleware: require role "user"
roleMiddleware-->gameRoutes: next()

gameRoutes->gameParamsMiddleware: gameParamsSchema.safeParse(req.params)

alt invalid gameId
    gameParamsMiddleware->errorHandler: next(BadRequest)
    errorHandler-->client: 400 Invalid gameId
else valid gameId
    gameParamsMiddleware-->gameRoutes: next()

    gameRoutes->gameController: getGameState(req, res, next)
    gameController->gameService: getGameState(gameId, userId)

    gameService->gameService: loadForParticipant(gameId, userId)

    gameService->gameRepository: findById(gameId)
    gameRepository->MySQL: SELECT game data
    MySQL-->gameRepository: persisted data | null

    gameRepository->gameRepository: restore domain Game
    gameRepository-->gameService: Game | undefined

    alt game not found
        gameService-->gameController: throw GameNotFound
        gameController->errorHandler: next(err)
        errorHandler-->client: 404 Game not found
    else game found
        gameService->gameService: find HumanPlayer associated with userId

        alt user is not participant
            gameService-->gameController: throw Forbidden
            gameController->errorHandler: next(err)
            errorHandler-->client: 403 Forbidden
        else user is participant
           gameService->gameService: build game state response
    gameService-->gameController: gameState
            gameController-->client: 200 OK + JSON
        end
    end
end
