title Sequence diagram - POST /game/:gameId/countermeasure

client->gameRoutes: POST /game/:gameId/countermeasure
gameRoutes->authMiddleware: verify JWT
authMiddleware-->gameRoutes: next()
gameRoutes->userRoleMiddleware: verify role
userRoleMiddleware-->gameRoutes: next()
gameRoutes->gameController: makeCountermeasure(req, res, next)
gameController->gameService: makeCountermeasure(gameId, userId, body)
gameService->gameRepository: getGame(gameId)
gameRepository->MySQL: retrieve game data
MySQL-->gameRepository: GameData | null
gameRepository-->gameService: Game | null

alt game not found
    gameService-->gameController: throw GameNotFound
    gameController->errorHandler: next(err)
    errorHandler-->client: 404 Game not found

else game found
    gameService->gameService: verify userId is a participant
    
    alt user is not participant
        gameService-->gameController: throw Forbidden
        gameController->errorHandler: next(err)
        errorHandler-->client: 403 Forbidden

    else user is participant
         gameService->GameEngine: executeCountermeasure(game, player, countermeasure)
        
            alt countermeasure not allowed
                GameEngine-->gameService: throw countermeasure error
                gameService-->gameController: throw error
                gameController->errorHandler: next(err)
                errorHandler-->client: error response
                
            else countermeasure executed
                GameEngine-->gameService: countermeasure result

                gameService->gameRepository: save updated game and action history
                gameRepository->MySQL: UPDATE game state + INSERT history
                MySQL-->gameRepository: saved
                gameRepository-->gameService: updated

                gameService-->gameController: countermeasure result
                gameController-->client: 200 OK + JSON
            end
        end
    end
end
