title Sequence diagram - GET /game/:gameId/history

client->gameRoutes: GET /game/:gameId/history

gameRoutes->authMiddleware: verify JWT
authMiddleware-->gameRoutes: next()

gameRoutes->roleMiddleware: require role "user"
roleMiddleware-->gameRoutes: next()

gameRoutes->gameParamsMiddleware: validate gameId

alt invalid gameId
    gameParamsMiddleware->errorHandler: next(BadRequest)
    errorHandler-->client: 400 Invalid gameId
else valid gameId
    gameParamsMiddleware-->gameRoutes: next()

    gameRoutes->gameController: getHistory(req, res, next)
    gameController->gameService: getGameHistory(gameId, userId)

    gameService->gameService: loadForParticipant(gameId, userId)

    gameService->gameRepository: findById(gameId)
    gameRepository->MySQL: SELECT game, participants, fleets and move history
    MySQL-->gameRepository: persisted data | null

    gameRepository->gameRepository: restore domain Game and MoveRecords
    gameRepository-->gameService: Game | undefined

    alt game not found
        gameService-->gameController: throw GameNotFound
        gameController->errorHandler: next(err)
        errorHandler-->client: 404 Game not found
    else user is not participant
        gameService-->gameController: throw Forbidden
        gameController->errorHandler: next(err)
        errorHandler-->client: 403 Forbidden
    else participant
        gameService->HistoryCsv: historyToCsv(game, player)

        HistoryCsv->GameDomain: getVisibleResult(record, viewer)
        GameDomain-->HistoryCsv: visible result or HIDDEN

        HistoryCsv->HistoryCsv: create CSV
        HistoryCsv-->gameService: CSV

        gameService-->gameController: CSV
        gameController-->client: 200 OK + Content-Type text/csv
    end
end
