title Sequence diagram - POST /auth/login

client->authRoutes: POST /auth/login
authRoutes->authController: login(req, res, next)
authController->authService: login(email, password)

authService->authRepository: findByEmail(email)
authRepository->MySQL: SELECT user WHERE email = ?
MySQL-->authRepository: User | null
authRepository-->authService: User | null

alt user not found
    authService-->authController: throw InvalidCredentials
    authController->errorHandler: next(err)
    errorHandler-->client: 401 Unauthorized
else user found
    authService->authService: bcrypt.compare(password, hash)

    alt password invalid
        authService-->authController: throw InvalidCredentials
        authController->errorHandler: next(err)
        errorHandler-->client: 401 Unauthorized
    else password valid
        authService->authService: jwt.sign(...)
        authService-->authController: token
        authController-->client: 200 OK + JWT
    end
end
