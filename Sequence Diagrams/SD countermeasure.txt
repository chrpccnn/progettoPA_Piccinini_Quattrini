title Sequence diagram - POST /game/:gameId/countermeasure

client->gameRoutes: POST /game/:gameId/countermeasure

gameRoutes->authMiddleware: verify JWT
authMiddleware-->gameRoutes: next()

gameRoutes->roleMiddleware: requireRole("user")
roleMiddleware-->gameRoutes: next()

gameRoutes->gameController: makeCountermeasure(req, res, next)
gameController->gameController: countermeasureSchema.safeParse(req.body)

alt invalid countermeasure body
    gameController->errorHandler: next(BadRequest)
    errorHandler-->client: 400 Invalid countermeasure data
end

gameController->gameService: performCountermeasure(gameId, userId, parsedBody)
gameService->gameService: begin transaction

gameService->gameRepository: findById(gameId, transaction, true)
gameRepository->MySQL: SELECT game and related data FOR UPDATE
MySQL-->gameRepository: persisted game data | null
gameRepository->gameRepository: restore domain Game
gameRepository-->gameService: Game | undefined

alt game not found
    gameService-->gameController: throw GameNotFound
    gameController->errorHandler: next(err)
    errorHandler-->client: 404 Game not found
end

gameService->gameService: verify userId is a participant

alt user is not participant
    gameService-->gameController: throw Forbidden
    gameController->errorHandler: next(err)
    errorHandler-->client: 403 Forbidden
end

gameService->gameService: buildCountermeasure(game, body)
gameService->gameService: use pending action as countermeasure target

gameService->GameEngine: executeCountermeasure(game, player, countermeasure)

alt countermeasure not allowed
    GameEngine-->gameService: throw countermeasure error
    gameService-->gameController: throw BadRequest
    gameController->errorHandler: next(err)
    errorHandler-->client: error response
end

GameEngine-->gameService: countermeasure result

gameService->gameRepository: update(game, transaction)
gameRepository->MySQL: UPDATE game, Energy, TOKEN, silence, pending and history
MySQL-->gameRepository: persisted
gameRepository-->gameService: updated

gameService->gameService: commit transaction
gameService-->gameController: countermeasure response
gameController-->client: 200 OK + JSON
